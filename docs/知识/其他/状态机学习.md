## 实际案例分析
以下是一个简单的C语言实现的状态机的例子，这个状态机模拟了一个简单的交通信号灯控制系统，它有三个状态：红灯、绿灯和黄灯，并且在一定时间内自动切换状态。

```c
#include <stdio.h>
#include <stdbool.h>

// 定义状态枚举类型
typedef enum {
    RED_LIGHT,
    GREEN_LIGHT,
    YELLOW_LIGHT
} TrafficLightState;

// 定义状态机结构体
typedef struct {
    TrafficLightState currentState;
    int timeRemaining; // 剩余切换时间（秒，这里仅作为示例）
} TrafficLightStateMachine;

// 状态转移函数
void transition(TrafficLightStateMachine *light) {
    static const int lightDuration[3] = {30, 45, 5}; // 红绿黄灯各自持续的时间（秒）

    // 减少剩余时间
    light->timeRemaining--;

    // 根据当前状态和剩余时间决定是否切换状态
    if (light->timeRemaining <= 0) {
        switch (light->currentState) {
            case RED_LIGHT:
                light->currentState = GREEN_LIGHT;
                light->timeRemaining = lightDuration[GREEN_LIGHT];
                printf("Traffic light changed to Green.\n");
                break;
            case GREEN_LIGHT:
                light->currentState = YELLOW_LIGHT;
                light->timeRemaining = lightDuration[YELLOW_LIGHT];
                printf("Traffic light changed to Yellow.\n");
                break;
            case YELLOW_LIGHT:
                light->currentState = RED_LIGHT;
                light->timeRemaining = lightDuration[RED_LIGHT];
                printf("Traffic light changed to Red.\n");
                break;
        }
    }
}

int main() {
    TrafficLightStateMachine trafficLight = {RED_LIGHT, 30}; // 初始化为红灯，持续时间为30秒
    
    while (true) {
        transition(&trafficLight); // 模拟每秒钟状态的检查和切换
        // 在实际应用中，这里可能会有延时或定时器来触发状态检查
    }

    return 0;
}
```

看完大概的内容，可以尝试自己实现一个状态机，比如一个简单的自动售货机，它有5个状态：空闲、等待支付、支付中、支付成功、支付失败，并且根据不同的状态，执行不同的操作。

```c
#include <stdio.h>

// 定义状态枚举类型
typedef enum {
    IDLE,
    WAIT_PAYMENT,
    PAYMENT_PROCESSING,
    PAYMENT_SUCCESS,
    PAYMENT_FAILED
} VendingMachineState;

// 定义状态机结构体
typedef struct {
    VendingMachineState currentState;
    bool paymentReceived; // 是否收到支付
} VendingMachine;

// 状态转移函数
void vendingMachineTransition(VendingMachine *machine, bool payment) {
    switch (machine->currentState) {
        case IDLE:
            if (payment) {
                machine->currentState = PAYMENT_PROCESSING;
                printf("Payment received, processing...\n");
            } else {
                printf("Please insert coins to start the transaction.\n");
            }
            break;
        case WAIT_PAYMENT:
            if (payment) {
                machine->currentState = PAYMENT_PROCESSING;
                printf("Waiting period over, now processing the payment...\n");
            }
            break;
        case PAYMENT_PROCESSING:
            // 这里假设一个非常简化的处理过程，实际中可能涉及到网络请求或硬件交互
            machine->paymentReceived = true; // 假设支付成功
            if (machine->paymentReceived) {
                machine->currentState = PAYMENT_SUCCESS;
                printf("Payment successful!\n");
            } else {
                machine->currentState = PAYMENT_FAILED;
                printf("Payment failed!\n");
            }
            break;
        case PAYMENT_SUCCESS:
            printf("Transaction completed successfully.\n");
            // 提供商品或者执行后续操作...
            machine->currentState = IDLE; // 结束交易后回到空闲状态
            break;
        case PAYMENT_FAILED:
            printf("Transaction failed, please try again.\n");
            machine->currentState = IDLE; // 支付失败后回到空闲状态
            break;
    }
}

int main() {
    VendingMachine vendingMachine = {IDLE, false};
    
    // 假设用户交互流程
    vendingMachineTransition(&vendingMachine, false); // 用户未投币，提示插入硬币
    vendingMachineTransition(&vendingMachine, true);  // 用户投币，开始支付流程
    vendingMachineTransition(&vendingMachine, false); // 不再接受额外投币，继续处理已接收到的支付
    // ...在此之后，状态机会自动根据支付结果转移到下一个状态

    return 0;
}
```

能感受到一点,和平时不同的是,平时写更注重于过程,精力在于找到状态转化的关键点,用代码表述出来,这种状态机的写法着眼于每个状态的转换,而把每个转换按当前状态分割,初看就觉得会有利于程序的维护和扩展。

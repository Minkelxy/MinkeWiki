<script>
const API = '/api/fitness';
let selectedFile = null;

function toggleForm(id) {
  const el = document.getElementById(id);
  el.style.display = el.style.display === 'block' ? 'none' : 'block';
}

function onFileSelected(e) {
  selectedFile = e.target.files[0];
  document.getElementById('upload-label').textContent = selectedFile ? selectedFile.name : '点击选择图片（JPG/PNG，最大10MB）';
}

async function uploadPhoto() {
  if (!selectedFile) { alert('请选择图片'); return; }
  const formData = new FormData();
  formData.append('file', selectedFile);
  const date = document.getElementById('p-date').value || new Date().toISOString().slice(0,10);
  const title = document.getElementById('p-title').value;
  const res = await fetch(API + '/photos?date=' + encodeURIComponent(date) + '&title=' + encodeURIComponent(title), {
    method: 'POST', body: formData
  });
  if (res.ok) {
    toggleForm('photo-form');
    selectedFile = null;
    document.getElementById('p-file').value = '';
    document.getElementById('p-title').value = '';
    document.getElementById('upload-label').textContent = '点击选择图片（JPG/PNG，最大10MB）';
    loadPhotos();
  } else {
    const err = await res.json();
    alert('上传失败: ' + (err.detail || '未知错误'));
  }
}

function showLightbox(src) {
  document.getElementById('lightbox-img').src = src;
  document.getElementById('lightbox').style.display = 'block';
}

async function deletePhoto(id) {
  if (!confirm('确定删除这张照片？')) return;
  await fetch(API + '/photos/' + id, { method: 'DELETE' });
  loadPhotos();
}

function formatSize(bytes) {
  if (bytes < 1024*1024) return (bytes/1024).toFixed(0) + 'KB';
  return (bytes/(1024*1024)).toFixed(1) + 'MB';
}

async function loadPhotos() {
  const res = await fetch(API + '/photos');
  const json = await res.json();
  document.getElementById('photo-count').textContent = '共 ' + json.total + ' 张照片';
  const grid = document.getElementById('photo-grid');
  grid.innerHTML = json.data.map(r =>
    `<div class="photo-card">
      <img src="${API}/photos/${r.path}" onclick="showLightbox(this.src)" loading="lazy">
      <div class="info">
        <strong>${r.title || '无标题'}</strong>
        <div class="date">${r.date} · ${formatSize(r.size)}</div>
      </div>
      <div class="actions">
        <a href="${API}/photos/${r.path}" download class="btn-sm" style="display:inline-block;text-decoration:none;color:#4051b5;border:1px solid #4051b5;border-radius:3px;">下载</a>
        <button class="btn-sm btn-danger" onclick="deletePhoto('${r.id}')">删除</button>
      </div>
    </div>`
  ).join('');
}

document.addEventListener('DOMContentLoaded', loadPhotos);
</script>

## 拍摄说明

### 拍摄要求

- **拍摄频率**: 每月1次(建议固定日期,如每月1号)
- **拍摄时间**: 早晨空腹,光线充足
- **拍摄角度**: 正面、侧面、背面
- **拍摄条件**: 
  - 相同光线条件
  - 相同背景
  - 相同姿势
  - 相同距离

### 拍摄姿势

**正面:**
- 双脚与肩同宽
- 双臂自然下垂或叉腰
- 目视前方
- 全身放松

**侧面:**
- 身体侧对镜头
- 双臂自然下垂
- 目视前方
- 展示身体轮廓

**背面:**
- 背对镜头
- 双臂自然下垂或叉腰
- 展示背部线条

### 文件命名规范

```
YYYY-MM-DD_部位_角度.jpg
例如: 2026-05-05_全身_正面.jpg
```

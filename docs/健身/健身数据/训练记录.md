<script>
// ============ 工具函数 ============
const API = '/api/fitness';

function toggleForm(id) {
  const el = document.getElementById(id);
  el.style.display = el.style.display === 'block' ? 'none' : 'block';
}

function switchTab(tab) {
  document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
  document.querySelectorAll('.tab')[tab === 'training' ? 0 : 1].classList.add('active');
  document.getElementById('tab-training').style.display = tab === 'training' ? 'block' : 'none';
  document.getElementById('tab-metrics').style.display = tab === 'metrics' ? 'block' : 'none';
  if (tab === 'metrics' && metricsChart) setTimeout(() => metricsChart.resize(), 50);
}

function formatVal(v) { return v != null && v !== '' ? v : '-'; }

// ============ 训练记录 ============
let exerciseCounter = 0;

function addExercise() {
  exerciseCounter++;
  const div = document.createElement('div');
  div.className = 'exercise-item';
  div.id = 'ex-' + exerciseCounter;
  div.innerHTML = `
    <div class="form-row">
      <label>动作<br><input type="text" id="ex-name-${exerciseCounter}" placeholder="杠铃卧推" style="width:140px;"></label>
      <label>组数<br><input type="number" id="ex-sets-${exerciseCounter}" style="width:60px;"></label>
      <label>次数<br><input type="text" id="ex-reps-${exerciseCounter}" placeholder="12,10,8,6" style="width:100px;"></label>
      <label>重量<br><input type="text" id="ex-weight-${exerciseCounter}" placeholder="40kg" style="width:80px;"></label>
      <label>RPE<br><input type="text" id="ex-rpe-${exerciseCounter}" placeholder="8" style="width:60px;"></label>
      <button class="btn-danger btn-sm" onclick="document.getElementById('ex-${exerciseCounter}').remove()">删除</button>
    </div>`;
  document.getElementById('exercise-list').appendChild(div);
}

function collectExercises() {
  const items = document.querySelectorAll('#exercise-list .exercise-item');
  const exercises = [];
  items.forEach(el => {
    const id = el.id.split('-')[1];
    const name = document.getElementById('ex-name-' + id)?.value;
    if (!name) return;
    exercises.push({
      name: name,
      sets: parseInt(document.getElementById('ex-sets-' + id)?.value) || null,
      reps: document.getElementById('ex-reps-' + id)?.value || '',
      weight: document.getElementById('ex-weight-' + id)?.value || '',
      rpe: document.getElementById('ex-rpe-' + id)?.value || ''
    });
  });
  return exercises;
}

async function saveTraining() {
  const exercises = collectExercises();
  if (exercises.length === 0) { alert('请至少添加一个动作'); return; }

  const record = {
    date: document.getElementById('t-date').value || new Date().toISOString().slice(0,10),
    training_day: document.getElementById('t-day').value,
    feeling: parseInt(document.getElementById('t-feeling').value) || null,
    exercises: exercises,
    cardio_type: document.getElementById('t-cardio-type').value,
    cardio_duration: parseInt(document.getElementById('t-cardio-dur').value) || null,
    notes: document.getElementById('t-notes').value
  };
  if (!record.date) { alert('请填写日期'); return; }

  const res = await fetch(API + '/training-records', {
    method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(record)
  });
  if (res.ok) {
    toggleForm('training-form');
    document.getElementById('exercise-list').innerHTML = '';
    loadTrainingRecords();
  }
}

async function deleteTraining(id) {
  if (!confirm('确定删除？')) return;
  await fetch(API + '/training-records/' + id, { method: 'DELETE' });
  loadTrainingRecords();
}

async function loadTrainingRecords() {
  const res = await fetch(API + '/training-records?limit=100');
  const json = await res.json();
  document.getElementById('training-count').textContent = `共 ${json.total} 条记录`;
  const tbody = document.querySelector('#training-table tbody');
  tbody.innerHTML = json.data.map(r => {
    const exs = r.exercises.map(e =>
      `<b>${e.name}</b> ${e.sets||''}×${e.reps||''} ${e.weight||''} RPE:${e.rpe||''}`
    ).join('<br>');
    const cardio = r.cardio_type ? `${r.cardio_type} ${r.cardio_duration||''}min` : '-';
    return `<tr>
      <td>${r.date}</td><td>${r.training_day}</td><td>${exs}</td>
      <td>${formatVal(r.feeling)}</td><td>${cardio}</td><td>${formatVal(r.notes)}</td>
      <td><button class="btn-danger btn-sm" onclick="deleteTraining('${r.id}')">删除</button></td>
    </tr>`;
  }).join('');
  drawTrainingChart(json.data);
}

// ============ 身体指标 ============
async function saveMetrics() {
  const metric = {
    date: document.getElementById('m-date').value || new Date().toISOString().slice(0,10),
    weight: parseFloat(document.getElementById('m-weight').value) || null,
    waist: parseFloat(document.getElementById('m-waist').value) || null,
    arm: parseFloat(document.getElementById('m-arm').value) || null,
    systolic_bp: parseInt(document.getElementById('m-sbp').value) || null,
    diastolic_bp: parseInt(document.getElementById('m-dbp').value) || null,
    bench_press: parseFloat(document.getElementById('m-bench').value) || null,
    squat: parseFloat(document.getElementById('m-squat').value) || null,
    deadlift: parseFloat(document.getElementById('m-deadlift').value) || null,
    pull_ups: parseInt(document.getElementById('m-pullups').value) || null,
    grip_strength: parseFloat(document.getElementById('m-grip').value) || null,
    plank: parseInt(document.getElementById('m-plank').value) || null,
    notes: document.getElementById('m-notes').value
  };
  const res = await fetch(API + '/body-metrics', {
    method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify(metric)
  });
  if (res.ok) {
    toggleForm('metrics-form');
    loadBodyMetrics();
  }
}

async function deleteMetric(id) {
  if (!confirm('确定删除？')) return;
  await fetch(API + '/body-metrics/' + id, { method: 'DELETE' });
  loadBodyMetrics();
}

async function loadBodyMetrics() {
  const res = await fetch(API + '/body-metrics?limit=100');
  const json = await res.json();
  const tbody = document.querySelector('#metrics-table tbody');
  const bp = (r.systolic_bp || r.diastolic_bp) ? `${formatVal(r.systolic_bp)}/${formatVal(r.diastolic_bp)}` : '-';
  const plankMin = r.plank ? `${Math.floor(r.plank/60)}'${r.plank%60}"` : '-';
  tbody.innerHTML = json.data.map(r =>
    `<tr>
      <td>${r.date}</td><td>${formatVal(r.weight)}</td><td>${formatVal(r.waist)}</td>
      <td>${formatVal(r.arm)}</td><td>${bp}</td>
      <td>${formatVal(r.bench_press)}</td><td>${formatVal(r.squat)}</td><td>${formatVal(r.deadlift)}</td>
      <td>${formatVal(r.pull_ups)}</td><td>${formatVal(r.grip_strength)}</td><td>${plankMin}</td>
      <td>${formatVal(r.notes)}</td>
      <td><button class="btn-danger btn-sm" onclick="deleteMetric('${r.id}')">删除</button></td>
    </tr>`
  ).join('');
  drawMetricsChart(json.data);
}

// ============ 导出 ============
function exportCSV(type) {
  window.open(API + '/export/' + (type === 'training' ? 'training' : 'body-metrics'), '_blank');
}

// ============ 初始化 ============
document.addEventListener('DOMContentLoaded', () => {
  loadTrainingRecords();
  loadBodyMetrics();
});

// ============ 趋势图表 ============
let trainingChart = null;
function drawTrainingChart(data) {
  const el = document.getElementById('training-chart');
  if (!el || typeof echarts === 'undefined') return;
  if (!trainingChart) trainingChart = echarts.init(el);
  const sorted = [...data].sort((a, b) => (a.date || '').localeCompare(b.date || ''));
  const totalSets = sorted.map(d => (d.exercises || []).reduce((s, e) => s + (parseInt(e.sets) || 0), 0));
  trainingChart.setOption({
    title: { text: '训练感受与训练量', left: 'center', textStyle: { fontSize: 14 } },
    tooltip: { trigger: 'axis' },
    legend: { data: ['感受 (1-10)', '总组数'], top: 26 },
    grid: { left: 45, right: 45, top: 55, bottom: 30 },
    xAxis: { type: 'category', data: sorted.map(d => d.date) },
    yAxis: [
      { type: 'value', min: 0, max: 10, name: '感受' },
      { type: 'value', min: 0, name: '组数' }
    ],
    series: [
      { name: '感受 (1-10)', type: 'line', data: sorted.map(d => d.feeling), smooth: true, connectNulls: true, areaStyle: { opacity: 0.15 } },
      { name: '总组数', type: 'bar', yAxisIndex: 1, data: totalSets, itemStyle: { color: 'rgba(33,150,243,0.4)' } }
    ]
  });
}

let metricsChart = null;
function drawMetricsChart(data) {
  const el = document.getElementById('metrics-chart');
  if (!el || typeof echarts === 'undefined') return;
  if (!metricsChart) metricsChart = echarts.init(el);
  const sorted = [...data].sort((a, b) => (a.date || '').localeCompare(b.date || ''));
  metricsChart.setOption({
    title: { text: '体重 / 腰围趋势', left: 'center', textStyle: { fontSize: 14 } },
    tooltip: { trigger: 'axis' },
    legend: { data: ['体重 (kg)', '腰围 (cm)'], top: 26 },
    grid: { left: 45, right: 20, top: 55, bottom: 30 },
    xAxis: { type: 'category', data: sorted.map(d => d.date) },
    yAxis: { type: 'value', scale: true },
    series: [
      { name: '体重 (kg)', type: 'line', data: sorted.map(d => d.weight), smooth: true, connectNulls: true },
      { name: '腰围 (cm)', type: 'line', data: sorted.map(d => d.waist), smooth: true, connectNulls: true }
    ]
  });
}
